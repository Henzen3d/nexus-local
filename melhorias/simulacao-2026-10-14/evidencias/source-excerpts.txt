69:     if user_id is None:

75: async def _current_role(user_id: str) -> Optional[str]:
76:     """Role atual do usuário no banco ou ``None`` se ele não existe mais."""
77:     now = time.monotonic()
78:     cached = _user_state_cache.get(user_id)
79:     if cached and cached[0] > now:
80:         return cached[1]
81:     async with _user_state_lock:
82:         cached = _user_state_cache.get(user_id)
83:         if cached and cached[0] > now:
84:             return cached[1]
85:         db = await get_db()
86:         try:
87:             async with db.execute("SELECT role FROM users WHERE id = ?", (user_id,)) as cur:
88:                 row = await cur.fetchone()
89:         finally:
90:             await db.close()
91:         role = (row[0] or "user") if row else None
92:         _user_state_cache[user_id] = (now + _USER_STATE_TTL_SECONDS, role)
93:         return role

96: def _decode_claims(token: str) -> Optional[dict]:
97:     try:
98:         payload = jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])
99:     except JWTError:
100:         return None
101:     user_id = payload.get("sub")
102:     username = payload.get("username")
103:     if not user_id or not username:
104:         return None
105:     return {"id": user_id, "username": username, "role": payload.get("role", "user")}

107:
108: async def get_current_user(credentials: HTTPAuthorizationCredentials = Depends(security)) -> dict:
109:     credentials_exception = HTTPException(
110:         status_code=status.HTTP_401_UNAUTHORIZED,
111:         detail="Token de autenticação inválido ou expirado.",
112:         headers={"WWW-Authenticate": "Bearer"},
113:     )
114:     user = _decode_claims(credentials.credentials)
115:     if not user:
116:         raise credentials_exception
117:     role = await _current_role(user["id"])
118:     if role is None:
119:         raise credentials_exception
120:     user["role"] = role
121:     return user
122:
123:
124: async def require_admin(user: dict = Depends(get_current_user)) -> dict:
125:     if user.get("role") != "admin":
126:         raise HTTPException(
127:             status_code=status.HTTP_403_FORBIDDEN,
128:             detail="Acesso restrito a administradores.",
129:         )
130:     return user

133:     """Decodifica o token sem consultar o banco (compatibilidade). Prefira ``resolve_token_user``."""
134:     return _decode_claims(token)

137: async def resolve_token_user(token: str) -> Optional[dict]:

139:     user = _decode_claims(token)

459:             if scope_prefix

462:         scope_params = (len(scope_prefix), scope_prefix) if scope_prefix else ()

465:         try:

469:                        WHERE model_id = ? AND user_id = ? AND fingerprint = ? AND created_at > ?

478:                             (exact_hit[0],),

480:                         await db.commit()

484:                             similarity=1.0,

486:                             token_est=exact_hit[2] or 0,
487:                         )
488:         except Exception as e:
489:             logger.debug("[cache_semantic] exact hash check bypass: %s", e)

491:         # JEV Fast Passo 2: Obter embedding da query limpa (reutiliza em 0ms do request_context)
492:         embed_target = (query_text or prompt_text).strip()

506:         # Load all candidates for this model

508:             async with db.execute(
509:                 f"""SELECT id, embedding, response, token_est FROM cache_semantic
510:                    WHERE model_id = ? AND user_id = ? AND created_at > ?

514:                 rows = await cur.fetchall()

517:             return None

520:         best_sim = 0.0

132: async def _refinar_pergunta(
133:     provider_juiz, judge_model_name: str, prompt: str, history: list
134: ) -> str:
135:     """Refina a pergunta do usuário para remover ambiguidades antes do fan-out."""
136:
137:     system_prompt = (
138:         "Você é um refinador de prompts. Seu objetivo é pegar a pergunta do usuário e reescrevê-la para que fique cristalina, "
139:         "resolvendo ambiguidades e completando referências implícitas (como 'ele', 'isso', 'anterior') com base no histórico da conversa.\n\n"
140:         "REGRAS:\n"
141:         "1. Clarifique sem prescrever: deixe explícito O QUE o usuário quer, mas NÃO prescreva COMO responder. Preserve a abertura para diferentes abordagens.\n"
142:         "2. EXTREMAMENTE IMPORTANTE: Só altere a pergunta se houver uma ambiguidade REAL dependente do histórico (ex: uso de pronomes ou continuidade de assunto). "
143:         "Se a pergunta já é compreensível e auto-contida, você DEVE retornar EXATAMENTE o texto original. Não tente 'melhorar', reescrever datas relativas ou adicionar colchetes didáticos à pergunta.\n"
144:         "3. Retorne APENAS o texto da pergunta refinada (ou original), sem comentários, aspas extras, ou explicações."
145:     )
146:
147:     # Só o histórico recente — evita prompt enorme (ex.: system de projeto) no refinador
148:     hist_curto = []
149:     for m in history:
150:         if not isinstance(m, dict):
151:             continue
152:         role = m.get("role")
153:         if role not in ("user", "assistant"):
154:             continue
155:         content = m.get("content")
156:         if isinstance(content, list):
157:             text_parts = [
158:                 str(p.get("text", "")) for p in content if isinstance(p, dict) and p.get("type") == "text"
159:             ]
160:             content = "\n".join(text_parts)
161:         if not isinstance(content, str) or not content.strip():
162:             continue
163:         # Cap por mensagem para não estourar free-tier no refine
164:         hist_curto.append({"role": role, "content": content[:2000]})
165:     hist_curto = hist_curto[-6:]
166:
167:     mensagens = [{"role": "system", "content": system_prompt}] + hist_curto
168:     if not any(m.get("role") == "user" for m in hist_curto):
169:         mensagens.append({"role": "user", "content": prompt[:4000]})
170:
171:     try:
172:         async def _stream_refine() -> str:
173:             resposta = ""
174:             async for token in provider_juiz.stream_chat(
175:                 model=judge_model_name,
176:                 messages=mensagens,
177:             ):
178:                 resposta += str(token)
179:             return resposta
180:
181:         resposta = await asyncio.wait_for(_stream_refine(), timeout=TIMEOUT_REFINAR)
182:         resposta_limpa = resposta.strip()
183:
184:         # Se a IA encapsular com aspas, removemos
185:         if resposta_limpa.startswith('"') and resposta_limpa.endswith('"'):
186:             resposta_limpa = resposta_limpa[1:-1].strip()
187:
188:         # Refino vazio ou absurdamente longo → fallback
189:         if not resposta_limpa or len(resposta_limpa) > max(len(prompt) * 4, 4000):
190:             return prompt
191:
192:         return resposta_limpa
193:     except asyncio.TimeoutError:
194:         logger.warning("fusion refinement: timeout após %ss — usando pergunta original", TIMEOUT_REFINAR)
195:         return prompt
196:     except Exception:
197:         logger.warning("fusion refinement: erro ao refinar pergunta", exc_info=True)
198:         return prompt

201: def _precisa_refinar(prompt: str, history: list) -> bool:
202:     """Heurística rápida: só aciona o refinador se houver sinais de ambiguidade.
203:     Suporta todas as línguas de tradução ativa no sistema (PT, EN, ES, FR, DE, JA, KO)."""
204:     # Primeira mensagem da conversa — não há contexto anterior para desambiguar
205:     user_msgs = [m for m in history if isinstance(m, dict) and m.get("role") == "user"]
206:     if len(user_msgs) <= 1:
207:         return False
208:
209:     # Padrões para línguas com limite de palavra (\b)
210:     # (Português, Inglês, Espanhol, Francês, Alemão)
211:     patterns_western = [
212:         # Português
213:         r'isso|isto|ele[s]?|ela[s]?|anterior|acima|aquilo|o mesmo|a mesma|o que disse|como mencionei|continue|continua|e quanto|sobre aquilo|nesse caso|desse jeito|esse tema|essa questão|do que falamos',
214:         # English
215:         r'this|that|it|he|she|they|him|her|them|above|former|latter|aforesaid|same|continue|go on|what about|about that|in this case|in that case|what I said|as I mentioned',
216:         # Español
217:         r'esto|eso|aquello|él|ella|ellos|ellas|anterior|arriba|lo mismo|la misma|continua|continúa|qué hay de|sobre eso|en este caso|en ese caso|lo que dije|como mencioné',
218:         # Français
219:         r'ceci|cela|ça|il|elle|ils|elles|le|la|les|lui|leur|précédent|ci-dessus|continuer|qu\'en est-il|à ce sujet|dans ce cas|ce que j\'ai dit|comme mentionné',
220:         # Deutsch
221:         r'dies|das|es|er|sie|ihm|ihn|ihnen|vorherige|oben|weiter|weitergehen|was ist mit|darüber|in diesem Fall|was ich gesagt habe|wie erwähnt'
222:     ]
223:
224:     combined_western = r'\b(' + '|'.join(patterns_western) + r')\b'
225:     if re.search(combined_western, prompt, re.IGNORECASE):
226:         return True
227:
228:     # Padrões para línguas sem limite de palavra (Japonês, Coreano)
229:     patterns_asian = [
230:         # Japanese (これ, それ, あれ, 彼, 彼女, 前の, 続き, 続けて, さっき)
231:         r'これ|それ|あれ|彼|彼女|前の|続き|続けて|さっき',
232:         # Korean (이것, 그것, 저것, 그, 그녀, 이전, 계속, 아까, 앞서)
233:         r'이것|그것|저것|그|그녀|이전|계속|아까|앞서'
234:     ]
235:     combined_asian = '|'.join(patterns_asian)
236:     if re.search(combined_asian, prompt):
237:         return True
238:
239:     return False

344:     # Substituir no history apenas para os modelos (o bd ainda guarda o original)
345:     history_para_modelos = [dict(m) for m in history]
346:     if history_para_modelos and history_para_modelos[-1].get("role") == "user":
347:         original_content = history_para_modelos[-1]["content"]
348:         if isinstance(original_content, str):
349:             history_para_modelos[-1]["content"] = original_content.replace(prompt, prompt_refinado)
350:         elif isinstance(original_content, list):
351:             # Caso seja multimodal (com imagens), substituímos na parte de texto

354:                     part["text"] = part["text"].replace(prompt, prompt_refinado)

426:     min_keep: int = 2,
427: ) -> tuple[list[dict], int]:
428:     """Mantém as mensagens mais recentes que cabem em ``budget_tokens``.

432:     - A última mensagem do usuário é SEMPRE mantida, mesmo que sozinha
433:       exceda o orçamento (o provedor devolve um erro claro nesse caso, em vez
434:       de a aplicação descartar silenciosamente a pergunta).
435:     - Remove sempre do início (mensagens mais antigas) para preservar a
436:       coerência do fim da conversa; nunca remove menos que ``min_keep`` do fim

93: def embedding_to_blob(vec) -> bytes:
94:     arr = np.array(vec, dtype=np.float32)
95:     return arr.tobytes()

97:
98: def blob_to_embedding(blob: bytes):
99:     return np.frombuffer(blob, dtype=np.float32)

102: def cosine_similarity(a, b) -> float:
103:     denom = float(np.linalg.norm(a) * np.linalg.norm(b))
104:     if denom == 0:
105:         return 0.0
106:     return float(np.dot(a, b) / denom)

109: def embed_text(text: str) -> Optional[bytes]:
110:     """Sync embed → BLOB float32. Returns None if embedder unavailable."""
111:     embedder = init_embedder()
112:     if embedder is None or not _HAS_NUMPY:
113:         return None
114:     if not text or not str(text).strip():
115:         return None
116:     embeddings = list(embedder.embed([str(text)[:8000]]))
117:     return embedding_to_blob(embeddings[0])
118:
119:
120: def embed_texts(texts: list[str]) -> list[Optional[bytes]]:
121:     embedder = init_embedder()
122:     if embedder is None or not _HAS_NUMPY or not texts:
123:         return [None] * len(texts)
124:     cleaned = [str(t)[:8000] if t else "" for t in texts]
125:     # Preserve empties as None without calling embed on empty batch only
126:     non_empty_idx = [i for i, t in enumerate(cleaned) if t.strip()]
127:     result: list[Optional[bytes]] = [None] * len(texts)
128:     if not non_empty_idx:
129:         return result
130:     batch = [cleaned[i] for i in non_empty_idx]
131:     vectors = list(embedder.embed(batch))
132:     for j, i in enumerate(non_empty_idx):
133:         result[i] = embedding_to_blob(vectors[j])
134:     return result

78:     context_window: int,

106:     # If model window is tiny, respect it

164: def _simple_keyword_score(query: str, text: str) -> float:
165:     """Cheap token overlap for fallback ranking (no embeddings)."""
166:     q = {t.lower() for t in re.findall(r"[a-zA-ZÀ-ÿ0-9_]{2,}", query or "")}
167:     if not q or not text:
168:         return 0.0
169:     d = {t.lower() for t in re.findall(r"[a-zA-ZÀ-ÿ0-9_]{2,}", text)}
170:     if not d:
171:         return 0.0
172:     return len(q & d) / len(q)

175: def fit_chunks_to_token_budget(
176:     chunks: list[dict],
177:     max_tokens: int,
178:     text_key: str = "chunk_text",
179: ) -> tuple[list[dict], bool]:
180:     """
181:     Greedy keep highest-score chunks until max_tokens is reached.
182:     Assumes chunks already sorted by score desc.
183:     Returns (selected, was_truncated).
184:     """
185:     if max_tokens <= 0 or not chunks:
186:         return [], bool(chunks)
187:
188:     selected: list[dict] = []
189:     used = 0
190:     for c in chunks:
191:         t = estimate_tokens(c.get(text_key) or "")
192:         # Always allow at least one chunk (truncated) if budget is tiny
193:         if selected and used + t > max_tokens:
194:             continue
195:         if not selected and t > max_tokens:
196:             # Truncate single oversized chunk on a line boundary
197:             text = c.get(text_key) or ""
198:             keep_chars = max(200, max_tokens * 4)
199:             cut = text[:keep_chars]
200:             nl = cut.rfind("\n")
201:             if nl > keep_chars // 2:
202:                 cut = cut[:nl]
203:             c = {**c, text_key: cut.rstrip() + "\n…[trecho limitado por orçamento de tokens]"}
204:             selected.append(c)
205:             return selected, True
206:         selected.append(c)
207:         used += t
208:     return selected, len(selected) < len(chunks)

79: def _lexical_score(query_tokens: list[str], doc_text: str) -> float:
80:     """
81:     Lightweight BM25-ish / coverage score in [0, 1].
82:     Rewards exact term hits and multi-term coverage without external deps.
83:     """
84:     if not query_tokens or not doc_text:
85:         return 0.0
86:     doc_tokens = _tokenize(doc_text)
87:     if not doc_tokens:
88:         return 0.0
89:
90:     doc_tf = Counter(doc_tokens)
91:     doc_len = len(doc_tokens)
92:     # Unique query terms (preserve multi-hit weight lightly via tf)
93:     q_unique = list(dict.fromkeys(query_tokens))
94:     if not q_unique:
95:         return 0.0
96:
97:     hit = 0.0
98:     covered = 0
99:     for t in q_unique:
100:         tf = doc_tf.get(t, 0)
101:         if tf <= 0:
102:             continue
103:         covered += 1
104:         # log tf saturation
105:         hit += 1.0 + math.log(1.0 + tf)
106:
107:     coverage = covered / len(q_unique)
108:     # Length-normalized density of hits
109:     density = hit / (1.0 + math.log(1.0 + doc_len / 40.0))
110:     # Blend coverage (main) with density, clamp
111:     score = 0.7 * coverage + 0.3 * min(1.0, density / max(1.0, len(q_unique)))
112:     return max(0.0, min(1.0, score))

95: async def _migrate_project_tags(db, user_id: str) -> int:
96:     """
97:     One-shot: convert legacy conversation.project_tag groups into real projects.
98:     - Creates a project per distinct non-empty tag (matched by name, case-insensitive)
99:     - Sets conversations.project_id for those chats
100:     Returns number of projects created.
101:     """
102:     created = 0
103:     try:
104:         async with db.execute(
105:             """SELECT DISTINCT TRIM(project_tag) AS tag
106:                FROM conversations
107:                WHERE user_id = ?
108:                  AND project_tag IS NOT NULL
109:                  AND TRIM(project_tag) != ''""",
110:             (user_id,),
111:         ) as cur:
112:             tags = [r[0] for r in await cur.fetchall() if r[0]]
113:     except Exception:
114:         return 0
115:
116:     for tag in tags:
117:         # Prefer exact name match; also try slug-ish forms
118:         async with db.execute(
119:             """SELECT id FROM projects
120:                WHERE user_id = ? AND archived = 0
121:                  AND (LOWER(name) = LOWER(?) OR LOWER(name) = LOWER(?))
122:                LIMIT 1""",
123:             (user_id, tag, tag.replace("-", " ")),
124:         ) as cur:
125:             existing = await cur.fetchone()
126:
127:         if existing:
128:             project_id = existing[0]
129:         else:
130:             project_id = str(uuid.uuid4())
131:             # Humanize slug: nicolas stays nicolas; fix-servicos → Fix Servicos-ish
132:             display = tag.replace("-", " ").strip()
133:             if display:
134:                 display = display[:1].upper() + display[1:]
135:             await db.execute(
136:                 """INSERT INTO projects (id, user_id, name, description)
137:                    VALUES (?, ?, ?, ?)""",
138:                 (project_id, user_id, display or tag, f"Migrado da tag «{tag}»"),
139:             )
140:             await db.execute(
141:                 """INSERT INTO project_memory (id, project_id, summary_text)
142:                    VALUES (?, ?, '')""",
143:                 (str(uuid.uuid4()), project_id),
144:             )
145:             created += 1
146:
147:         await db.execute(
148:             """UPDATE conversations
149:                SET project_id = ?
150:                WHERE user_id = ?
151:                  AND project_id IS NULL
152:                  AND project_tag IS NOT NULL
153:                  AND TRIM(project_tag) = ?""",
154:             (project_id, user_id, tag),
155:         )
156:
157:     if tags:
158:         await db.commit()
159:     return created

271:           let dropText = i18n.t('chat.effortDroppedUnsupported', {
272:             defaultValue: 'O modelo selecionado não suporta controle de esforço.',
273:           })
274:           if (msg.reason === 'failover') {
275:             dropText = i18n.t('chat.effortDroppedFailover', {
276:               defaultValue: 'O modelo reserva não suporta controle de esforço; resposta gerada no padrão.',
277:             })
278:           } else if (msg.reason === 'provider_rejected') {
279:             dropText = i18n.t('chat.effortDroppedProvider', {
280:               defaultValue: 'O provedor rejeitou o parâmetro de esforço; retentando sem ele.',
281:             })
282:           }
283:           s.showToast(dropText, 'info')
284:           break
285:         }
286:         case 'error':
287:           if (!isGeneratingRef.current) return
288:           isGeneratingRef.current = false
289:           console.error('[WS error]', msg.message)
290:           s.setStreaming(false)
291:           s.setFusionActive(false)
292:           s.showToast(msg.message, 'error')
293:           s.addErrorMessage(msg.message)
294:           break
295:       }
296:     }
297:
298:     const scheduleReconnect = (reason: string) => {
299:       if (disposed) return
300:       clearRetry()
301:       const delay = Math.min(MAX_RETRY_MS, BASE_RETRY_MS * 2 ** attempt)
302:       attempt += 1
303:       if (attempt <= 3 || attempt % 5 === 0) {
304:         console.warn(`[WS] reconnect in ${delay}ms (attempt ${attempt}): ${reason}`)
305:       }
306:       retryTimer = setTimeout(connect, delay)
307:     }
308:
309:     const connect = () => {
310:       if (disposed) return
311:
312:       // Evita sockets órfãos se connect for chamado enquanto ainda CONNECTING/OPEN.
313:       if (
314:         socket &&
315:         (socket.readyState === WebSocket.OPEN || socket.readyState === WebSocket.CONNECTING)
316:       ) {
317:         return
318:       }
319:
320:       let ws: WebSocket
321:       try {
322:         ws = new WebSocket(buildWsUrl(token))
323:       } catch (err) {
324:         console.error('[WS] failed to construct socket', err)
325:         scheduleReconnect('constructor error')
326:         return
327:       }
328:
329:       socket = ws
330:       wsRef.current = ws
331:
332:       ws.onopen = () => {
333:         attempt = 0
334:         startHeartbeat(ws)
335:         // Presença é reenviada ao receber `ready` (socket registrado no backend)
336:       }
337:
338:       ws.onmessage = handleMessage
339:
340:       ws.onerror = () => {
341:         // onclose always follows; keep logging light to avoid console spam.
342:         if (attempt === 0) {
343:           console.warn('[WS] connection error')
344:         }
345:       }
346:
347:       ws.onclose = (ev) => {
348:         clearHeartbeat()
349:         isGeneratingRef.current = false
350:         if (wsRef.current === ws) wsRef.current = null
351:         if (socket === ws) socket = null
352:
353:         // Nunca deixar a UI presa em "Fusion ativo" / streaming se o socket caiu
354:         const st = useStore.getState()
355:         if (st.fusionActive || st.isStreaming) {
356:           st.setStreaming(false)
357:           st.setFusionActive(false)
358:           st.showToast(
359:             i18n.t('chat.wsDroppedDuringStream', {
360:               defaultValue: 'Conexão interrompida durante a resposta. Tente novamente.',
361:             }),
362:             'error'
363:           )
364:         }
365:
366:         if (disposed) return
367:
368:         // Auth reject: não adianta martelar reconexão com o mesmo token inválido.
369:         if (ev.code === AUTH_CLOSE_CODE) {
370:           console.warn('[WS] auth rejected — faça login novamente')
371:           useStore.getState().showToast(
372:             i18n.t('chat.wsAuthFailed', { defaultValue: 'Sessão expirada. Faça login novamente.' }),
373:             'error'
374:           )
375:           return
376:         }
377:
378:         scheduleReconnect(`code=${ev.code} reason=${ev.reason || 'closed'}`)
379:       }
380:     }
381:
382:     connect()
383:
384:     // Reconecta ao voltar online ou ao focar a aba (útil após sleep/túnel Cloudflare)
385:     const onOnline = () => {
386:       if (disposed) return
387:       if (!wsRef.current || wsRef.current.readyState !== WebSocket.OPEN) {
388:         attempt = 0
389:         clearRetry()
390:         connect()
391:       }
392:     }
393:     const onVisibility = () => {
394:       if (document.visibilityState === 'visible') onOnline()
395:     }
396:     window.addEventListener('online', onOnline)
397:     document.addEventListener('visibilitychange', onVisibility)
398:
399:     return () => {
400:       disposed = true
401:       clearRetry()
402:       clearHeartbeat()
403:       window.removeEventListener('online', onOnline)
404:       document.removeEventListener('visibilitychange', onVisibility)
405:       // Fecha sem disparar reconnect (flag disposed).
406:       if (socket) {
407:         socket.onclose = null
408:         socket.onerror = null
409:         socket.onmessage = null
410:         socket.onopen = null
411:         try {
412:           socket.close()
413:         } catch {
414:           /* ignore */
415:         }
416:       }
417:       socket = null
418:       if (wsRef.current) wsRef.current = null
419:     }
420:   }, [token, resendPresence])

422:   const stopStream = useCallback(() => {

424:     const s = useStore.getState()

426:
